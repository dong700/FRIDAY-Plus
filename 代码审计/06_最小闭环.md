## 闭环 1：WebSocket 一问一答（对话骨架）

**问题**：前端发 `chat.send`，后端流式回 `mind.delta`，最后 `mind.end`。怎么保证前端不会看到半截标记、不会重复渲染？

**最小实现**（需要 `pip install fastapi uvicorn`）：

```python
# min_ws.py
import json, asyncio
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
import uvicorn

app = FastAPI()

async def fake_llm(text: str):
    """假装是流式 LLM：逐字吐。"""
    for ch in f"你说的是「{text}」，我收到了。":
        await asyncio.sleep(0.02)
        yield ch

@app.websocket("/ws")
async def ws(sock: WebSocket):
    await sock.accept()
    try:
        while True:
            ev = json.loads(await sock.receive_text())
            if ev.get("type") != "chat.send":
                continue
            mid = "m1"
            await sock.send_text(json.dumps({"type": "mind.start", "payload": {"id": mid}}))
            buf = []
            async for ch in fake_llm(ev["payload"]["content"]):
                buf.append(ch)
                await sock.send_text(json.dumps(
                    {"type": "mind.delta", "payload": {"id": mid, "text": ch}},
                    ensure_ascii=False))
            await sock.send_text(json.dumps(
                {"type": "mind.end", "payload": {"id": mid, "text": "".join(buf)}},
                ensure_ascii=False))
    except WebSocketDisconnect:
        pass

if __name__ == "__main__":
    uvicorn.run(app, host="127.0.0.1", port=18080, ws="wsproto")
```

配套客户端（`min_ws_client.py`）：

```python
import asyncio, json, websockets

async def main():
    async with websockets.connect("ws://127.0.0.1:18080/ws") as ws:
        await ws.send(json.dumps({"type": "chat.send", "payload": {"content": "在吗"}}))
        while True:
            ev = json.loads(await ws.recv())
            if ev["type"] == "mind.delta":
                print(ev["payload"]["text"], end="", flush=True)
            elif ev["type"] == "mind.end":
                print("\n--- end ---"); break

asyncio.run(main())
```

**与源码对应**
- `api/server.py` `websocket_endpoint`（477–500）：同样的 `while True: receive → handle → send`。
- 正式版多了 `chat.message` 广播（让所有前端都看到）和 `avatar.*`（让立绘动），但那不是"闭环"的必要部分。

**最小版故意没做**
- 没有 `manager.broadcast`（单连接不需要）
- 没有 `client_msg_id` 去重（前端才需要）
- 没有异常回包

**部署坑**
1. `uvicorn.run(ws="wsproto")` 不能省：websockets 16 + 旧 uvicorn 握手会 403。
2. 客户端如果不读 `mind.end` 就 `break`，服务端会收到 `WebSocketDisconnect`，日志里一条无害的堆栈。
3. 中文字符串一定要 `ensure_ascii=False`，否则前端收到 `\u4f60` 也能显示，但抓包排查时极难读。

## 闭环 2：提醒（说一句 → 到点叫 → 关闭）

**问题**：用户说"10 分钟后提醒我喝水"，系统要能：解析出时间、存下来、到点被叫醒、说出口后关闭。

**最小实现**（纯标准库）：

```python
# min_reminder.py
import re, time, json, threading
from datetime import datetime, timedelta
from pathlib import Path

WEEKDAY = {"一": 0, "二": 1, "三": 2, "四": 3, "五": 4, "六": 5, "日": 6, "天": 6}
TRIGGER = ("提醒我", "提醒一下", "别忘了", "别忘", "帮我记", "记住")
CLOCK_WORDS = ("下午", "晚上", "今晚", "傍晚", "中午", "早上", "上午")

def parse_when(raw: str):
    now = datetime.now()
    # 分钟最具体，必须放在「天/小时」之前
    m = re.search(r"(\d+)\s*(?:分钟|分)\s*(?:之后|以后|后)", raw)
    if m:
        return (now + timedelta(minutes=int(m.group(1)))).timestamp()
    m = re.search(r"(\d+)\s*(?:小时|钟头)\s*(?:之后|以后|后)", raw)
    if m:
        return (now + timedelta(hours=int(m.group(1)))).timestamp()
    m = re.search(r"(\d+)\s*(?:天|日)\s*(?:之后|以后|后)", raw)
    if m:
        return (now + timedelta(days=int(m.group(1)))).timestamp()

    base = None
    if "大后天" in raw: base = now + timedelta(days=3)
    elif "后天" in raw: base = now + timedelta(days=2)
    elif "明天" in raw: base = now + timedelta(days=1)
    elif "今天" in raw or "今晚" in raw: base = now

    m = re.search(r"(下)?(?:周|星期)([一二三四五六日天])", raw)
    if m:
        t = WEEKDAY[m.group(2)]
        ahead = (7 - now.weekday()) + t if m.group(1) else (t - now.weekday()) % 7
        base = (now + timedelta(days=ahead)).replace(hour=9, minute=0, second=0, microsecond=0)

    if base is None:
        # 只说「晚上」「三点」没给日期 → 按今天算；纯「帮我记住…」仍返回 None
        if not (any(k in raw for k in CLOCK_WORDS)
                or re.search(r"(\d{1,2})\s*[点:：]", raw)
                or re.search(r"[一二两三四五六七八九十]{1,3}\s*[点時时]", raw)):
            return None
        base = now

    hh, mm = base.hour, base.minute
    got_clock = False
    tm = re.search(r"(\d{1,2})\s*[点:：]\s*(\d{1,2})?", raw)
    if tm:
        hh, mm, got_clock = int(tm.group(1)), int(tm.group(2) or 0), True
    if got_clock:
        if ("下午" in raw or "晚上" in raw or "今晚" in raw) and hh < 12:
            hh += 12
    else:
        mm = 0                      # 修复点：分钟必须清零
        if "下午" in raw: hh = 14
        elif "晚上" in raw or "今晚" in raw: hh = 20
        elif "中午" in raw: hh = 12
        else: hh = 9                # 无时刻线索 → 09:00

    dt = base.replace(hour=hh, minute=mm, second=0, microsecond=0)
    if dt <= now:
        dt += timedelta(days=1)
    return dt.timestamp()

class Reminders:
    def __init__(self, path="reminders.jsonl"):
        self.path = Path(path)
        self.items = []
        if self.path.exists():
            self.items = [json.loads(l) for l in self.path.read_text("utf-8").splitlines() if l.strip()]

    def _save(self):
        self.path.write_text("\n".join(json.dumps(x, ensure_ascii=False) for x in self.items), "utf-8")

    def maybe_add(self, text: str):
        if not any(k in text for k in TRIGGER):
            return None
        due = parse_when(text)
        if due is None and "提醒" not in text:
            return None
        # 同文去重
        for it in self.items:
            if it["text"] == text[:200] and not it.get("done"):
                return it
        it = {"id": f"r{len(self.items)+1}", "text": text[:200], "due_at": due, "done": False}
        self.items.append(it); self._save(); return it

    def due(self):
        now = time.time()
        return [x for x in self.items
                if x.get("due_at") and not x["done"] and x["due_at"] <= now]

    def mark_done(self, rid):
        for x in self.items:
            if x["id"] == rid:
                x["done"] = True
        self._save()

if __name__ == "__main__":
    r = Reminders("/tmp/rem.jsonl") if False else Reminders("rem_demo.jsonl")
    print(r.maybe_add("10分钟后提醒我喝水"))
    it = r.maybe_add("明天下午三点提醒我交报告")
    print(it, datetime.fromtimestamp(it["due_at"]) if it["due_at"] else None)
```

**与源码对应**
- `_parse_when`（`server.py` 2841）：本版只保留"分钟/小时/明天/周X/钟点"，去掉了"月日/N号/三刻"。
- `TFGR.remember(due_at=...)` / `due()` / `mark_done()`（`tfgr.py` 498/571/557）。

**最小版故意没做**
- 没有 `pin`、`importance`、`last_hit`
- 没有"回复命中 40% 就关闭"的 `_reply_mentions`
- 没有 6 小时防重复（`_due_notified`）

**部署坑**
1. **`datetime.now()` 无时区**。跨时区/容器时区不对，提醒会整体偏移。端到端跑一次 `原理库/02_reminder.py` 回归：`今晚`/`晚上`/`10分钟后`/`帮我记住` 四条必须分别落到 20:00 / 今天 20:00 / +10 分钟 / 只钉住。
2. **分钟必须在无钟点分支清零**（曾经 bug）：`hh, mm = base.hour, base.minute` 只改 `hh`，"今晚" 会落在"现在这一分"。已修。
3. **"三点"歧义**：下午 4 点说"三点" → 明天 03:00，用户要的是"明天下午三点"。仍未处理（需要上下文/时段推断）。
4. **`Path.write_text` 非原子**：进程在写一半时被杀 → 文件损坏 → 下次启动 `json.loads` 抛错。正式版用 `_safe_write_json` 就是为了这个。

## 闭环 3：长期记忆的"挑"（importance + digest）

**问题**：有 800 条记忆，这一轮只给模型 10 条。挑哪 10 条？

**最小实现**（纯标准库）：

```python
# min_digest.py
import time, difflib

KIND_W = {"memory": 0.9, "intent": 0.7, "belief": 0.6, "pattern": 0.55, "story": 0.5}
PIN, DUE, HALF, NEG = 1.0, 1.5, 14.0, 0.55

class E:
    def __init__(self, id, kind, text, ts=None, conf=1.0, pin=False, due=None):
        self.id, self.kind, self.text = id, kind, text
        self.ts = ts or time.time()
        self.conf = conf
        self.pin, self.due = pin, due
        self.last_hit = self.ts
        self.negated = False

def importance(e: E, now=None):
    now = now or time.time()
    s = KIND_W.get(e.kind, 0.35)
    s += 0.25 * min(1.0, max(0.0, e.conf))
    if e.pin: s += PIN
    if e.due: s += DUE
    age = max(0.0, (now - e.last_hit) / 86400)
    s *= 0.5 ** (age / HALF)
    if e.negated: s *= NEG
    return round(s, 3)

def overlap(q, t):
    qs = {c for c in q if "\u4e00" <= c <= "\u9fff"}
    if not qs: return 0.0
    t = t.lower()
    return sum(1 for c in qs if c in t) / len(qs)

def digest(items, query="", max_items=10, max_chars=800):
    now = time.time()
    shown, picked = [], []

    def take(e, dedup=False):
        if len(picked) >= max_items or e in picked: return
        if dedup and any(difflib.SequenceMatcher(None, e.text, s).ratio() >= 0.75 for s in shown):
            return
        picked.append(e); shown.append(e.text)

    def score(e):
        s = importance(e, now) + 0.6 * overlap(query, e.text)
        return s

    def rank(e): return (-score(e), -e.ts)

    # 层1 到点
    for e in sorted([x for x in items if x.due and not getattr(x, "done", False)], key=lambda x: x.due):
        take(e)
    # 层2 用户让记
    for e in sorted([x for x in items if x.kind == "memory"], key=rank): take(e)
    # 层3 钉住的推断
    for e in sorted([x for x in items if x.pin and x.kind != "memory"], key=rank): take(e, dedup=True)
    # 层4 其余推断
    for e in sorted([x for x in items if x.kind in KIND_W], key=rank): take(e, dedup=True)

    out, used = [], 0
    for e in picked:
        e.last_hit = now
        tag = "他让记" if e.kind == "memory" else "我推断"
        line = f"{tag}·{e.text[:60]}"
        if used + len(line) > max_chars and out: break
        out.append(line); used += len(line)
    return "\n".join(out)

if __name__ == "__main__":
    now = time.time()
    items = [
        E("1", "memory", "我不吃辣", ts=now-5*86400, pin=True),
        E("2", "belief", "他大概在赶项目，周末也没休", ts=now-3*86400, conf=0.6),
        E("3", "belief", "他喜欢猫", ts=now-40*86400, conf=0.8),
        E("4", "memory", "周五交房租", ts=now-1*86400, pin=True, due=now+2*86400),
        E("5", "pattern", "他每天 9 点开始写代码", ts=now-10*86400, conf=0.7),
    ]
    print(digest(items, query="房租的事"))
```

**输出预期**：`周五交房租` 排第一（due 加成 + query 命中），然后 `我不吃辣`（memory + pin），再是 belief。

**与源码对应**：`tfgr.py` 的 `importance`（423）、`digest`（590）、`_query_overlap`（108）、`_near_dup`（122）。

**最小版故意没做**
- 没有跨来源去重（`exclude_texts`）
- 没有 10 分钟粘性
- 没有字符预算的精确截断（本版是整行丢弃）

**部署坑**
1. **`last_hit` 在 `take` 时更新** → 本版没注入的也会被标记吗？不会，本版只在输出循环里更新（比正式版安全，正式版在 `break` 前就更新了）。
2. **`overlap` 按单字**，"我/的/他"会虚抬所有条目。
3. **`sorted(..., key=rank)` 每次重算 `score`**，800 条 × 3 次排序 = 2400 次 `overlap`，可感知的卡顿。

## 闭环 4：负反馈（用户否定 → 降权）

```python
# min_negate.py
import re, time

# 与源码 tfgr.py 的 NEGATION_RE 完全一致（注意：不含单独的「不对」，只含「不对吧」）
NEG = re.compile(r"你记错|记错了|我没说过|没说过|不是这样|别乱说|胡说|搞错|不对吧|没有这回事")

def penalize(items, user_text, window=1800, factor=0.6):
    if not NEG.search(user_text):
        return []
    now = time.time()
    cands = [e for e in items
             if getattr(e, "last_inject", 0) and now - e.last_inject <= window]
    if not cands:
        return []
    cands.sort(key=lambda e: -e.last_inject)
    hit = [e for e in cands if overlap(user_text, e.text) >= 0.3]
    target = hit[:3] if hit else cands[:2]
    for e in target:
        e.conf = max(0.1, round(e.conf * factor, 2))
        e.negated = True
    return [e.id for e in target]

def overlap(q, t):
    qs = {c for c in q if "\u4e00" <= c <= "\u9fff"}
    return sum(1 for c in qs if c in t) / len(qs) if qs else 0.0
```

**三种情况的实测输出**（`原理库/04_negate.py`）：

```
精准否定 "你记错了，我周末没赶项目"  -> ['1']          （只罚字面重合的那条）
泛否定   "你记错了"                  -> ['4', '5']    （罚最近注入的 2 条）
纯泛否定 "不对"                      -> []            （正则不命中，不动）
```

**与源码对应**：`penalize_recent`（463）。

**部署坑**
1. **"不对"也是转折词**（"不对，你刚才那句挺好"）→ 误伤。
2. **永久降权不可恢复**：三次否定后置信度 0.1，这条事实再也浮不上来。正式版也没解决。

## 闭环 5：心智循环（事件 → 触发 → 五道闸 → 说）

**问题**：系统要"自己醒来说话"，但不能抢麦、不能复读、不能打断工作。

**最小实现**（纯标准库，用线程模拟循环）：

```python
# min_mind.py
import time, threading, difflib, re

def norm(t): return re.sub(r"[^\u4e00-\u9fffa-z0-9]+", "", t.lower())

TEMPLATE_PREFIX_RE = re.compile(
    r"^(?:提醒你一下|提醒一下你|提醒一下|顺便说一句|顺便说一下|对了|那个)[：:，,\s]*")

def dedup_key(t):
    """剥掉固定模板前缀再查重，避免同模板的主动消息互相误杀。"""
    s = (t or "").strip()
    return TEMPLATE_PREFIX_RE.sub("", s).strip() or s

def too_similar(a, b):
    na, nb = norm(a), norm(b)
    if not na or not nb: return False
    if difflib.SequenceMatcher(None, na, nb).ratio() >= 0.62: return True
    ga = {na[i:i+2] for i in range(len(na)-1)}
    gb = {nb[i:i+2] for i in range(len(nb)-1)}
    return bool(ga and gb and len(ga & gb)/min(len(ga), len(gb)) >= 0.55)

class Mind:
    def __init__(self, say):
        self.say = say                  # 回调：真正发消息
        self.hooks = []                 # [(due_ts, note)]
        self.last_says = []
        self.ignored = 0
        self.in_work = False
        self.last_activity = time.time()
        self._stop = False

    def add_hook(self, seconds, note):
        self.hooks.append((time.time() + seconds, note))

    def user_said(self):
        self.ignored = 0
        self.last_activity = time.time()

    def on_user_ignored(self):
        self.ignored += 1

    def should_speak(self, text):
        if not text.strip(): return False, "空"
        if re.fullmatch(r"[（(][^）)]*[）)]", text.strip()): return False, "旁白式沉默"
        for prev in self.last_says[-5:]:
            if too_similar(dedup_key(text), dedup_key(prev)): return False, "和上一句太像"
        if self.ignored >= 2: return False, "连续两次没回应"
        if self.in_work: return False, "他在工作"
        return True, "ok"

    def loop(self):
        while not self._stop:
            now = time.time()
            fires = [h for h in self.hooks if h[0] <= now]
            if fires:
                self.hooks = [h for h in self.hooks if h[0] > now]
                text = fires[0][1]
                ok, why = self.should_speak(text)
                print(f"[mind] 想说话（{why}）: {text}")
                if ok:
                    self.say(text); self.last_says.append(text)
            time.sleep(0.3)

if __name__ == "__main__":
    m = Mind(say=lambda t: print(f"[气泡] {t}"))
    threading.Thread(target=m.loop, daemon=True).start()

    m.add_hook(0.6, "该喝水了，你已经两小时没起身")
    m.add_hook(0.9, "三点要开会，记得带电脑")      # 文案不同 → 能说

    m.in_work = True                                # 闸 5
    m.add_hook(0.3, "晚饭想吃点什么")
    time.sleep(1.6)

    m.in_work, m.ignored = False, 2                 # 闸 4
    m.add_hook(0.2, "要不要一起看个视频")
    time.sleep(0.6)

    m.user_said()                                   # 用户开口 → 闸门恢复
    m.add_hook(0.2, "要不要一起看个视频")
    time.sleep(0.9)
```

**实测输出**（`原理库/05_mind.py`）：

```
[气泡] 该喝水了，你已经两小时没起身
[气泡] 三点要开会，记得带电脑
[mind] 想说（他在工作软件里）: 晚饭想吃点什么
[mind] 想说（连续两次主动说话没被回应）: 要不要一起看个视频
[气泡] 要不要一起看个视频
```

**与源码对应**：`_should_speak`（808）、`_too_similar`（608）、`_dedup_key`（601）、`_select_trigger`（1030）、`_next_timeout`（1009）。

**最小版故意没做**
- 没有优先级（正式版：到点 > 钩子 > 事件）
- 没有 `idle` 钩子
- 没有压话（`_defer_say`）

**部署坑**
1. **`time.sleep(0.5)` 在独立线程里**是安全的；如果这是 asyncio 协程，必须 `await asyncio.sleep`。
2. **`ignored` 只有 `user_said` 能重置**；用户"在看但没打字"会被当成无视。
3. **`should_speak` 没有"工作软件检测"**，本版用 `in_work` 布尔占位。
4. **固定模板会互相误杀**（曾经 bug）：`提醒你一下：喝水 / 吃药` 相似度 0.875 → 第二条直接不说。已用 `dedup_key` 剥前缀修复，回归见 `原理库/05_mind.py` 最后两行。

## 闭环 6：ATG 活动合并（同一件事别切碎）

**问题**：视觉每 15 秒给一句"在用电脑"，不能每句都开一个新活动。

```python
# min_atg.py
import time, difflib

class Act:
    def __init__(self, desc, task="", t=None):
        self.id = f"A{id(self)%10000}"
        self.desc, self.task = desc, task
        self.t0 = t or time.time()
        self.t1 = None
        self.paused = False
        self.interrupt_by = None
        self.resume_of = None

    def duration(self, now=None):
        return (self.t1 or now or time.time()) - self.t0

class ATG:
    def __init__(self, merge=0.6, gap=30.0, timeout=600.0):
        self.acts = []
        self.merge, self.gap, self.timeout = merge, gap, timeout

    def current(self):
        return next((a for a in reversed(self.acts) if a.t1 is None and not a.paused), None)

    def observe(self, desc, task="", now=None):
        now = now or time.time()
        cur = self.current()
        if cur:
            ratio = difflib.SequenceMatcher(None, cur.desc, desc).ratio()
            if ratio >= self.merge:
                cur.t1 = None                       # 继续延长
                if task and not cur.task:
                    cur.task = task
                return cur.id, "merge"
            if now - cur.t0 < self.gap:
                # 时间太近、描述不同 → 并发，不新建
                return cur.id, "overlap"
            cur.paused = True
            cur.t1 = now
        # 看有没有能恢复的
        for a in reversed(self.acts):
            if a.paused and difflib.SequenceMatcher(None, a.desc, desc).ratio() >= self.merge:
                a.paused = False; a.t1 = None; a.resume_of = a.id
                return a.id, "resume"
        a = Act(desc, task, now)
        self.acts.append(a)
        return a.id, "new"

if __name__ == "__main__":
    g = ATG(gap=30.0)
    t = time.time()
    print(g.observe("在写代码", "实现登录", t))     # new
    print(g.observe("在用电脑", "", t + 10))       # 10s 内、描述不同 → overlap
    print(g.observe("在写代码", "", t + 40))       # 描述相同 → merge
    print(g.observe("在吃饭", "", t + 80))         # 距起点 80s → 打断
    print(g.observe("在写代码", "", t + 200))      # 命中 paused → resume
    print([(a.desc, a.paused, a.resume_of) for a in g.acts])
```

**实测输出**（`原理库/06_atg.py`）：

```
('A0001', 'new')
('A0001', 'overlap')
('A0001', 'merge')
('A0002', 'new')
('A0001', 'resume')
```

**与源码对应**：`ActivityTimeline.observe_activity`、`merge_similarity_threshold=0.6`、`interrupt_gap_threshold=30`、`auto_complete_timeout_s=120`。

**最小版故意没做**
- 没有图结构（正式版用 `networkx.MultiDiGraph`）
- 没有 `ApplicaionEdgeType`（BEFORE/OVERLAPS/INTERRUPT…）
- 没有 `context_snapshot`、`entities`、`confidence`

**部署坑**
1. **`merge_similarity_threshold=0.6` 用字面相似度**："在写代码" vs "在调试代码" 约 0.5 → 不合并 → 碎片。
2. **`auto_complete_timeout_s=120`** 太短：静坐 10 分钟看代码 → 活动被自动完成 → 再开一个。
3. **`id(self)%10000`** 会撞（最小版偷懒）；正式版用递增计数器 + 锁。

## 闭环 7：语义检索（embedding + 字面重排）

**问题**：本地 embedding 在中文短句上排序不稳，纯向量召回经常把不相关的排前面。

```python
# min_semantic.py  （需要 pip install ollama 或用 requests 直接打 11434）
import json, math, urllib.request

def embed(text):
    req = urllib.request.Request(
        "http://127.0.0.1:11434/api/embeddings",
        data=json.dumps({"model": "nomic-embed-text", "prompt": text}).encode(),
        headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=10) as r:
        return json.loads(r.read())["embedding"]

def cosine(a, b):
    dot = sum(x*y for x, y in zip(a, b))
    na = math.sqrt(sum(x*x for x in a)); nb = math.sqrt(sum(y*y for y in b))
    return dot / (na*nb) if na and nb else 0.0

def overlap(q, t):
    qs = {c for c in q if "\u4e00" <= c <= "\u9fff"}
    return sum(1 for c in qs if c in t) / len(qs) if qs else 0.0

def find(query, docs, k=3, min_score=0.55):
    qv = embed(query)
    scored = []
    for d in docs:
        s = cosine(qv, d["vec"])
        s = 0.7*s + 0.3*overlap(query, d["text"])   # 字面重排
        scored.append((s, d))
    scored.sort(key=lambda x: -x[0])
    return [(round(s, 3), d["text"]) for s, d in scored[:k] if s >= min_score]

if __name__ == "__main__":
    texts = ["周五要交房租", "我不吃辣", "他最近在赶一个项目", "他喜欢猫"]
    docs = [{"text": t, "vec": embed(t)} for t in texts]
    for s, t in find("房租的事", docs):
        print(s, t)
```

**与源码对应**：`ov_memory.find`（99）、`_ov_label`（`server.py` 2945）负责把 URI 翻成 `[11-03 我推断]`。

**最小版故意没做**
- 没有持久化（每次启动重算）
- 没有 OpenViking 服务端
- 没有两级沉淀（单条 + 按天汇总）

**部署坑**
1. **Ollama 没起 → `urlopen` 抛 URLError**，必须 catch 并降级为空（正式版就是这么做的，代价是"它突然变笨了"但没人知道）。
2. **维度写死 768**：换成 1024 维的模型，旧向量全部失效。
3. **`min_score=0.55`** 是经验值；中文短句的余弦普遍偏高（0.6+），阈值太松会召回噪音。

## 闭环 8：工具调用（模型说"调工具"，你执行）

**问题**：LLM 返回 `tool_calls`，你要执行并回填结果，模型才能基于结果说话。

```python
# min_toolcall.py （伪 LLM 版本，不依赖网络）
import json

TOOLS = {
    "get_disk_free": lambda: {"free_gb": 123.4},
    "open_app": lambda name: {"opened": name},
}

SCHEMA = [
    {"type": "function", "function": {
        "name": "get_disk_free", "description": "查磁盘剩余空间",
        "parameters": {"type": "object", "properties": {}}}},
    {"type": "function", "function": {
        "name": "open_app", "description": "打开一个应用",
        "parameters": {"type": "object",
                       "properties": {"name": {"type": "string"}},
                       "required": ["name"]}}},
]

def fake_model_call(user_text):
    """假装模型决定调用工具。"""
    if "磁盘" in user_text:
        return {"content": "", "tool_calls": [
            {"id": "c1", "name": "get_disk_free", "arguments": {}}]}
    if "打开" in user_text:
        return {"content": "", "tool_calls": [
            {"id": "c2", "name": "open_app", "arguments": {"name": "记事本"}}]}
    return {"content": "我不需要工具", "tool_calls": []}

def run_turn(user_text):
    r = fake_model_call(user_text)
    if not r["tool_calls"]:
        return r["content"]
    results = []
    for call in r["tool_calls"]:
        fn = TOOLS.get(call["name"])
        if not fn:
            results.append({"id": call["id"], "error": f"未知工具 {call['name']}"})
            continue
        try:
            results.append({"id": call["id"], "result": fn(**call["arguments"])})
        except Exception as e:
            results.append({"id": call["id"], "error": str(e)})
    # 真实场景：把 results 作为 role=tool 消息再调一次模型
    return f"工具结果: {json.dumps(results, ensure_ascii=False)}"

if __name__ == "__main__":
    print(run_turn("磁盘还剩多少"))
    print(run_turn("打开记事本"))
    print(run_turn("你好"))
```

**与源码对应**：`llm.chat_with_tools`（141）解析 `tool_calls`；`BrainTools._tools` 是执行表；`handle_chat_send` 里机械臂那段（1054+）就是真实用法。

**最小版故意没做**
- 没有第二轮模型调用（真实要 `role:"tool"` 回填）
- 没有超时、没有并发
- 没有参数校验（schema 只是给模型看的，执行前必须再校验）

**部署坑**
1. **`json.loads(arguments)` 可能失败**（模型输出截断）→ 必须 try。
2. **schema 和 `TOOLS` 键不同步** → `KeyError`。
3. **工具执行卡住** → 整轮对话卡住（正式版没有统一超时）。
4. **模型可能编造工具名** → 必须判 `TOOLS.get`。

## 闭环 9：原子写（Windows 上别用 NamedTemporaryFile）

```python
# min_atomic.py
import json, os, uuid, time
from pathlib import Path

def safe_write(path: Path, data, retries=5):
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = json.dumps(data, ensure_ascii=False, indent=2)
    last = None
    for i in range(retries):
        tmp = f"{path}.{uuid.uuid4().hex[:8]}.tmp"
        try:
            with open(tmp, "w", encoding="utf-8") as f:
                f.write(payload)
            os.replace(tmp, path)          # 同分区原子替换
            return
        except Exception as e:
            last = e
            try: os.unlink(tmp)
            except Exception: pass
            if i < retries - 1:
                time.sleep(0.1 * (2 ** i))
    raise last

def read(path: Path, default=None):
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return default
```

**与源码对应**：`_safe_write_json`（`server.py` 390）。

**这个最小版比正式版好在哪**：`time.sleep` 在这里是安全的，因为它是**同步函数**；正式版把它放进 async 路由里就是 bug（阻塞事件循环）。**同一个函数，调用位置决定它是不是 bug**——这是审查里最容易被忽略的一类。

**部署坑**
1. `raise last` 在 `retries=0` 时是 `raise None` → `TypeError`。
2. 临时文件残留（进程被杀）：`*.tmp` 会一直堆着，需要启动时清理。
3. 跨分区（`os.replace` 从 C: 到 D:）**不是原子**，会退化成复制 + 删除。

## 闭环 10：流式过滤（把 `<memory>` 和钩子从气泡里抠掉）

**问题**：模型边生成边推送，但 `<memory>...</memory>` 和 `<<after=120 note=...>>` 不能让用户看到，而且标记可能正好被切成两个 chunk。

```python
# min_stream_filter.py
M_OPEN, M_CLOSE = "<<", ">>"
MEM_OPEN, MEM_CLOSE = "<memory>", "</memory>"

class StreamFilter:
    def __init__(self):
        self.buf = ""
        self.markers, self.memory = [], ""
        self.in_mem = False

    def _hold(self):
        best = 0
        for mk in (M_OPEN, MEM_OPEN):
            for k in range(len(mk)-1, 0, -1):
                if self.buf.endswith(mk[:k]):
                    best = max(best, k); break
        return best

    def feed(self, chunk):
        self.buf += chunk
        out = []
        while True:
            if self.in_mem:
                j = self.buf.find(MEM_CLOSE)
                if j < 0:
                    hold = len(MEM_CLOSE) - 1
                    if len(self.buf) > hold:
                        self.memory += self.buf[:-hold]
                        self.buf = self.buf[-hold:]
                    break
                self.memory += self.buf[:j]
                self.buf = self.buf[j+len(MEM_CLOSE):]
                self.in_mem = False
                continue
            i, m = self.buf.find(M_OPEN), self.buf.find(MEM_OPEN)
            if m >= 0 and (i < 0 or m < i):
                out.append(self.buf[:m]); self.buf = self.buf[m+len(MEM_OPEN):]
                self.in_mem = True; continue
            if i < 0:
                h = self._hold()
                if h:
                    out.append(self.buf[:-h]); self.buf = self.buf[-h:]
                else:
                    out.append(self.buf); self.buf = ""
                break
            j = self.buf.find(M_CLOSE, i+2)
            if j < 0:
                out.append(self.buf[:i]); self.buf = self.buf[i:]; break
            out.append(self.buf[:i])
            self.markers.append(self.buf[i+2:j])
            self.buf = self.buf[j+2:]
        return "".join(out)

    def finish(self):
        rest, self.buf = self.buf, ""
        if self.in_mem:
            return ""
        if M_OPEN in rest:
            rest = rest[:rest.find(M_OPEN)]
        return rest

if __name__ == "__main__":
    f = StreamFilter()
    chunks = ["你好", "，我记一下", "<memo", "ry>{\"a\":1}", "</memory>", " <<after=", "120 note=x>>", "好的"]
    print("".join(f.feed(c) for c in chunks))
    print("finish:", repr(f.finish()))
    print("memory:", f.memory, "markers:", f.markers)
```

**预期输出**：`你好，我记一下 好的`；`memory` 是 `{"a":1}`；`markers` 是 `["after=120 note=x"]`。

**与源码对应**：`StreamFilter`（`mind.py` 110）、`parse_hook_specs`（187）。

**部署坑**
1. **`finish()` 必须调用**，否则缓冲区里最后几个字符永远不显示（比如"好的"刚好被 hold 住）。
2. **未闭合的 `<memory>` 全部丢弃**：模型忘了写 `</memory>`，这段内容既不出气泡也不进库（静默丢失）。
3. **标记内嵌标记**（`<<a=1 <<b=2>>`）会被解析成一个 marker，第二个丢失。
